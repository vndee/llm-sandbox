Reproduction recipe (Debian 13 amd64, fresh scratch directory)
------------------------------------------------------------
The following is the successful build recipe, with dependency versions made
explicit. It uses the environment's inherited HTTPS proxy and CA trust. Network
commands require the execution tool's network permission. It does not require
root or install packages into the host. The base host must already provide Git,
GCC/G++, make, pkg-config, dpkg-deb, and the runtime packages listed below.

Run under bash:

  nsjail_build_root=/workspace/scratch/nsjail-build
  mkdir -p "$nsjail_build_root"
  git clone --no-checkout https://github.com/google/nsjail.git "$nsjail_build_root/source"
  git -C "$nsjail_build_root/source" checkout 4ff54a6e0d5b65a0e1633d6e2fd1425ba6c882ce
  git -C "$nsjail_build_root/source" submodule update --init --recursive
  test "$(git -C "$nsjail_build_root/source/kafel" rev-parse HEAD)" = 76d0f41bf3eb5c4008713d64b9767b461a9129a3

  mkdir -p "$nsjail_build_root/apt/lists/partial" \
    "$nsjail_build_root/apt/cache/archives/partial" \
    "$nsjail_build_root/apt/empty" "$nsjail_build_root/debs" "$nsjail_build_root/root"
  printf '%s\n' 'deb https://deb.debian.org/debian trixie main' > "$nsjail_build_root/apt/sources.list"
  apt-get \
    -o Dir::Etc::main=/dev/null \
    -o Dir::Etc::parts="$nsjail_build_root/apt/empty" \
    -o Dir::Etc::sourcelist="$nsjail_build_root/apt/sources.list" \
    -o Dir::Etc::sourceparts="$nsjail_build_root/apt/empty" \
    -o Dir::State::lists="$nsjail_build_root/apt/lists" \
    -o Dir::Cache="$nsjail_build_root/apt/cache" update
  cd "$nsjail_build_root/debs"
  apt-get \
    -o Dir::State::lists="$nsjail_build_root/apt/lists" \
    -o Dir::Etc::sourcelist="$nsjail_build_root/apt/sources.list" \
    -o Dir::Etc::sourceparts="$nsjail_build_root/apt/empty" \
    download \
    'bison=2:3.8.2+dfsg-1+b2' 'flex=2.6.4-8.2+b4' 'm4=1.4.19-8' \
    'protobuf-compiler=3.21.12-11+deb13u1' 'libprotobuf-dev=3.21.12-11+deb13u1' \
    'libprotoc32t64=3.21.12-11+deb13u1' 'libprotobuf32t64=3.21.12-11+deb13u1' \
    'libprotobuf-lite32t64=3.21.12-11+deb13u1' \
    'libnl-3-dev=3.7.0-2' 'libnl-route-3-dev=3.7.0-2'

  # Verify downloaded package hashes against the manifest below before extraction.
  for nsjail_deb in "$nsjail_build_root"/debs/*.deb; do
    dpkg-deb -x "$nsjail_deb" "$nsjail_build_root/root"
  done
  # Development packages contain relative .so symlinks. Supply their matching
  # runtime targets from the already installed host package versions listed below.
  cp /usr/lib/x86_64-linux-gnu/libnl-3.so.200.26.0 \
    /usr/lib/x86_64-linux-gnu/libnl-route-3.so.200.26.0 \
    "$nsjail_build_root/root/usr/lib/x86_64-linux-gnu/"
  cd "$nsjail_build_root/source"
  PATH="$nsjail_build_root/root/usr/bin:$PATH" \
    LD_LIBRARY_PATH="$nsjail_build_root/root/usr/lib/x86_64-linux-gnu" \
    PKG_CONFIG_PATH="$nsjail_build_root/root/usr/lib/x86_64-linux-gnu/pkgconfig" \
    PKG_CONFIG_SYSROOT_DIR="$nsjail_build_root/root" \
    BISON_PKGDATADIR="$nsjail_build_root/root/usr/share/bison" \
    M4="$nsjail_build_root/root/usr/bin/m4" \
    make -j2 USER_DEFINES=-DPR_SCHED_CORE_SCOPE_THREAD_GROUP=1 \
    > "$nsjail_build_root/build.log" 2>&1
  ldd "$nsjail_build_root/source/nsjail"
  sha256sum "$nsjail_build_root/source/nsjail"
